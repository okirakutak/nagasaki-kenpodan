api = EStatAPI(eStatAppId)
medicalExpensesDf = api.get_dataframe("0003356095")

# "時間軸（年度次）"を"年度"に変更
medicalExpensesDf = medicalExpensesDf.rename(columns={"時間軸（年度次）": "年度"})

# 年度の"年度"を外す
medicalExpensesDf["年度"] = medicalExpensesDf["年度"].str.replace("年度", "")

# 年度の型をint64に変換
medicalExpensesDf["年度"] = medicalExpensesDf["年度"].astype("int64")

# "都道府県"が"全国"でない、かつ"国民医療費・人口一人当たり国民医療費、診療種類"が"人口一人当たり国民医療費"の行を抜き出す
medicalExpensesDf = medicalExpensesDf[(medicalExpensesDf["都道府県"] != "全国") & (medicalExpensesDf["国民医療費・人口一人当たり国民医療費、診療種類"] == "人口一人当たり国民医療費")]

# "国民医療費・人口一人当たり国民医療費、診療種類"と"単位"を削除
medicalExpensesDf = medicalExpensesDf.drop(columns={"国民医療費・人口一人当たり国民医療費、診療種類", "単位"})

# "値"を"医療費[億円]"に変更
medicalExpensesDf = medicalExpensesDf.rename(columns={"値": "人口一人当たり国民医療費[千円]"})

# 数値型に変換
medicalExpensesDf["人口一人当たり国民医療費[千円]"] = pd.to_numeric(medicalExpensesDf["人口一人当たり国民医療費[千円]"], errors='coerce')

print(medicalExpensesDf)

from scipy import stats

# 保険料率と可処分所得のP値を計算
x = df['所定内給与額[千円]'].dropna()
y = df['人口一人当たり国民医療費[千円]'].dropna()

# 両方に値がある行のみ使用
valid = df[['所定内給与額[千円]', '人口一人当たり国民医療費[千円]']].dropna()
slope, intercept, r_value, p_value, std_err = stats.linregress(valid['所定内給与額[千円]'], valid['人口一人当たり国民医療費[千円]'])

print(f"相関係数 (r): {r_value:.4f}")
print(f"決定係数 (R²): {r_value**2:.4f}")
print(f"P値: {p_value:.4f}")
print(f"傾き: {slope:.4f}")
print(f"切片: {intercept:.4f}")
print()
if p_value < 0.05:
    print(f"P値 = {p_value:.4f} < 0.05 → 統計的に有意な相関あり")
else:
    print(f"P値 = {p_value:.4f} ≥ 0.05 → 統計的に有意な相関なし")
