一般化加法モデル（GAM）で、特定健診受診率と生産性の非線形な関係を確認する¶

線形回帰では「1%増えると平均的にどれだけ増えるか」を見ました。ここではGAMを使い、他のKPIを調整したうえで、特定健診受診率の効果が直線ではなく曲線的に変化しているかを確認します。

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from scipy.sparse import csc_matrix, csr_matrix
from pygam import LinearGAM, s

# pygam 0.9.x は SciPy の疎行列 .A 属性を参照するため、
# 新しい SciPy でも動くように互換プロパティを補います。
for sparse_cls in (csr_matrix, csc_matrix):
    if not hasattr(sparse_cls, 'A'):
        sparse_cls.A = property(lambda self: self.toarray())

gam_feature_cols = [
    '特定健診等',
    '特定保健指導',
    '受診勧奨',
    '後発医薬品',
    '対象者減少率',
]

gam_df = df[gam_feature_cols + ['県民一人あたり県内総生産[千円/人]']].dropna()
X_gam = gam_df[gam_feature_cols].to_numpy()
y_gam = gam_df['県民一人あたり県内総生産[千円/人]'].to_numpy()

terms = s(0, n_splines=6)
for i in range(1, len(gam_feature_cols)):
    terms += s(i, n_splines=6)

gam = LinearGAM(terms).gridsearch(
    X_gam,
    y_gam,
    lam=np.logspace(-3, 3, 7),
    progress=False,
)

pseudo_r2 = gam.statistics_['pseudo_r2']['explained_deviance']
print('【GAM結果】')
print(f"  Explained deviance: {pseudo_r2:.4f}")
print(f"  AIC: {gam.statistics_['AIC']:.2f}")
print(f"  有効自由度: {gam.statistics_['edof']:.2f}")
print(f"  選択された lambda: {np.asarray(gam.lam).ravel()[0]:.4g}")

# --- ここから修正部分 ---
# term=0（特定健診等）のグリッドを作成
XX = gam.generate_X_grid(term=0, n=100)

# 他のKPIを中央値に固定した「フル特徴量」グリッドを作成し、
# partial_dependence ではなく predict で絶対値スケールの予測曲線を得る
median_row = gam_df[gam_feature_cols].median().to_frame().T
XX_full = pd.concat([median_row] * len(XX), ignore_index=True)
XX_full['特定健診等'] = XX[:, 0]  # term=0 のグリッド値だけを差し替え

curve_pred = gam.predict(XX_full.to_numpy())
curve_ci = gam.confidence_intervals(XX_full.to_numpy(), width=0.95)
# --- 修正部分ここまで ---

fig, ax = plt.subplots(figsize=(8, 5))

# 予測カーブ（絶対値スケール、他のKPIは中央値固定）
ax.plot(XX[:, 0], curve_pred, color='#d62728', linewidth=2, label='GAM予測（他KPIは中央値固定）')
ax.fill_between(XX[:, 0], curve_ci[:, 0], curve_ci[:, 1], color='#d62728', alpha=0.18, label='95%信頼区間')

# 実測値（生のデータ点。他のKPIの値はバラバラなので、カーブからは外れて散らばる）
ax.scatter(

東京都を外した場合の感度分析とモデル比較¶

東京都は県民一人あたり県内総生産が突出して高いため、全体の関係が東京都に強く引っ張られていないかを確認します。ここでは、東京都を含む場合・外した場合について、重回帰分析とGAMを同じ説明変数で比較します。

# 東京都を含む場合/外した場合 × 重回帰/GAM の比較表
from IPython.display import display

TARGET_COL = '県民一人あたり県内総生産[千円/人]'
TOKYO_NAME = '東京'

comparison_feature_cols = [
    '特定健診等',
    '特定保健指導',
    '受診勧奨',
    '後発医薬品',
    '対象者減少率',
]


def fit_multiple_regression(data, label):
    analysis_df = data[comparison_feature_cols + [TARGET_COL]].dropna()
    X = sm.add_constant(analysis_df[comparison_feature_cols])
    y = analysis_df[TARGET_COL]
    model = sm.OLS(y, X).fit()
    return {
        '東京都': label,
        'モデル': '重回帰分析',
        'n': len(analysis_df),
        '適合指標': 'Adj. R²',
        '適合度': model.rsquared_adj,
        'AIC': model.aic,
        '特定健診等の効果': model.params['特定健診等'],
        '特定健診等 p値': model.pvalues['特定健診等'],
        '効果の解釈': '係数: 1pt増あたりの差',
    }


def fit_gam_comparison(data, label):
    analysis_df = data[comparison_feature_cols + [TARGET_COL]].dropna()
    X = analysis_df[comparison_feature_cols].to_numpy()
    y = analysis_df[TARGET_COL].to_numpy()

    terms = s(0, n_splines=6)
    for i in range(1, len(comparison_feature_cols)):
        terms += s(i, n_splines=6)

    model = LinearGAM(terms).gridsearch(
        X,
        y,
        lam=np.logspace(-3, 3, 7),
        progress=False,
    )

    # 他のKPIを中央値に固定し、特定健診等だけを10→90パーセンタイルに動かした予測差を効果量にします。
    rates = np.percentile(analysis_df['特定健診等'], [10, 90])
    median_row = analysis_df[comparison_feature_cols].median().to_frame().T
    check_X = pd.concat([median_row] * len(rates), ignore_index=True)
    check_X['特定健診等'] = rates
    pred = model.predict(check_X.to_numpy())

    return {
        '東京都': label,
        'モデル': 'GAM',
        'n': len(analysis_df),
        '適合指標': 'Explained deviance',
        '適合度': model.statistics_['pseudo_r2']['explained_deviance'],
        'AIC': model.statistics_['AIC'],
        '特定健診等の効果': pred[1] - pred[0],
        '特定健診等 p値': model.statistics_['p_values'][0],
        '効果の解釈': '10→90パーセンタイル予測差',
    }


datasets = {
    '含む': df.copy(),
    '外す': df[df['支部名'] != TOKYO_NAME].copy(),
}

comparison_rows = []
for tokyo_label, data_for_analysis in datasets.items():

]

比較表の読み方¶
